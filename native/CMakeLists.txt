cmake_minimum_required(VERSION 3.22)
project(kepyur C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(KEPYUR_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)
option(KEPYUR_RADIO "Build the radio (wifi-sdr + devourer + libusb)" ON)

# ---- DSP: no hardware, shared by the app and the desktop tools ----
add_library(kepyur_dsp STATIC dsp/video.cpp dsp/spectrum.cpp dsp/log.cpp)
target_include_directories(kepyur_dsp PUBLIC dsp)
target_compile_options(kepyur_dsp PRIVATE -O3 -ffast-math -Wall -Wextra)
if(ANDROID)
  target_link_libraries(kepyur_dsp PUBLIC log)
endif()

# ---- radio ----
if(KEPYUR_RADIO)
  if(ANDROID)
    include(cmake/libusb_android.cmake)  # libusb from source + a pkg-config file devourer can find
  endif()
  # wifi-sdr assembles the on-chip streamer with mipsel binutils (shell.nix: KEPYUR_MIPS_PREFIX)
  if(DEFINED ENV{KEPYUR_MIPS_PREFIX})
    set(MIPS_AS "$ENV{KEPYUR_MIPS_PREFIX}as" CACHE FILEPATH "")
    set(MIPS_LD "$ENV{KEPYUR_MIPS_PREFIX}ld" CACHE FILEPATH "")
    set(MIPS_OBJCOPY "$ENV{KEPYUR_MIPS_PREFIX}objcopy" CACHE FILEPATH "")
  endif()
  add_subdirectory(${KEPYUR_ROOT}/third_party/wifi-sdr wifi-sdr EXCLUDE_FROM_ALL)
  if(ANDROID)
    target_link_libraries(devourer PUBLIC usb-1.0-static)
  endif()
  add_library(kepyur_radio STATIC radio/radio.cpp)
  target_include_directories(kepyur_radio PUBLIC radio)
  target_link_libraries(kepyur_radio PUBLIC kepyur_dsp wsdrcore)
  target_compile_options(kepyur_radio PRIVATE -O2 -Wall -Wextra)
endif()

# ---- desktop tools ----
if(NOT ANDROID)
  find_package(Threads REQUIRED)
  add_executable(kp_replay tools/kp_replay.cpp)
  target_link_libraries(kp_replay PRIVATE kepyur_dsp)
  target_compile_options(kp_replay PRIVATE -O2 -Wall)
  if(KEPYUR_RADIO)
    add_executable(kp_live tools/kp_live.cpp)
    target_link_libraries(kp_live PRIVATE kepyur_radio Threads::Threads)
    target_compile_options(kp_live PRIVATE -O2 -Wall)
  endif()
endif()

# ---- Android JNI ----
if(ANDROID)
  add_library(kepyur SHARED jni/kepyur_jni.cpp)
  target_link_libraries(kepyur PRIVATE kepyur_radio android log)
  target_compile_options(kepyur PRIVATE -O2 -Wall)
endif()
